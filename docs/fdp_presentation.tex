% SLIDE 3: MULTI-AGENT SUPERVISOR ARCHITECTURE
% ==========================================
\begin{frame}{Core Pillar 1: Multi-Agent Supervisor Orchestration}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{The Supervisor Orchestration Pattern:}
      \begin{itemize}
        \item Instead of relying on a monolithic prompt, requests route to a centralized \alert{Orchestrator Agent} (\texttt{lib/agents/orchestrator.ts}).
        \item Coordinates execution across 4 specialized sub-agents:
      \end{itemize}
      
      \vspace{0.2cm}
      \small
      \begin{enumerate}
        \item \textbf{Resume Analyst Agent}: Scores ATS compatibility, skill gaps, and bullet impact.
        \item \textbf{Interview Coach Agent}: Generates STAR questions \& evaluates answers ($temp=0.1$).
        \item \textbf{RAG Grounding Agent}: Fetches domain context from vector memory.
        \item \textbf{Quality Critic Agent}: Evaluates output consistency using fast inference (\texttt{llama-3.1-8b-instant}).
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Pipeline Latency Bounding}
        \small
        \begin{itemize}
          \item \textbf{Parallel Tool Execution}: RAG vector retrieval \& MCP tool calls execute concurrently via \texttt{Promise.all()}.
          \item \textbf{Per-Agent Timeouts}: 8s upper bound per agent step with fallback heuristics.
          \item \textbf{Total Pipeline Budget}: Hard limit at $\sim 12$ seconds for complete analysis.
          \item \textbf{Stage Progress Streaming}: Pushes execution trace timeline updates to UI in real time.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ==========================================
% SLIDE 4: RETRIEVAL-AUGMENTED GENERATION (RAG)
% ==========================================
\begin{frame}{Core Pillar 2: In-Memory Retrieval-Augmented Generation (RAG)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{In-Memory Vector Store Architecture:}
      \begin{itemize}
        \item Powered by \texttt{MemoryVectorStore} and a custom \texttt{LocalEmbeddings} TF-IDF vectorizer (\texttt{lib/rag/embeddings.ts}).
        \item \textbf{Zero Infrastructure Overhead}: Eliminates external vector database dependencies (Pinecone/Chroma).
        \item Uses a 384-dimensional technical/career vocabulary derived from 4 curated JSON domain datasets:
          \begin{itemize}
            \item \texttt{interview-questions.json} (40+ questions)
            \item \texttt{bullet-rewrites.json} (30+ transformations)
            \item \texttt{star-examples.json} (15+ exemplary answers)
            \item \texttt{industry-keywords.json} (100+ skill taxonomy)
          \end{itemize}
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Inline RAG Grounding \& Explorer}
        \small
        \begin{itemize}
          \item \textbf{Inline Citations}: Actionable suggestions include expandable \texttt{GroundingCitation} badges displaying source snippet \& similarity score.
          \item \textbf{Knowledge Explorer Tab}: Integrated in \texttt{/dashboard?tab=knowledge} to search, filter, and inspect vector chunks.
          \item \textbf{Dynamic Vector Ingestion}: User-uploaded resume text and job descriptions are chunked (\texttt{RecursiveCharacterTextSplitter}) and ingested on-the-fly.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{lstlisting}[caption={Orchestrator Agent Parallel Pipeline}]
export class OrchestratorAgent extends BaseAgent {
  async executeTask(task: AgentTask, state: AgentState) {
    switch (task) {
      case 'analyze_resume':
        // Stage 1: Document Ingestion
        state = await ragAgent.ingestUserDocs(state);
        
        // Stage 2: Parallel RAG & MCP Execution
        const [ragRes, mcpRes] = await Promise.all([
          ragAgent.execute(state),
          mcpClient.executeTool('web_search', { query: state.jobDescription })
        ]);
        
        // Stage 3: Resume Scoring
        state = await resumeAgent.execute(state);
        
        // Stage 4: Quality Reflection Review
        return await criticAgent.reviewAnalysis(state);
    }
  }
}
\end{lstlisting}
    \end{column}

    \begin{column}{0.5\textwidth}
\begin{lstlisting}[caption={Deterministic Interview Scoring (temp=0.1)}]
export class BaseAgent {
  protected client: OpenAI;
  constructor(model?: string) {
    this.client = new OpenAI({
      baseURL: 'https://api.groq.com/openai/v1',
      apiKey: process.env.GROQ_API_KEY,
    });
  }
  protected async chat(messages, options) {
    return await this.client.chat.completions.create({
      model: this.model,
      messages,
      temperature: 0.1, // Deterministic evaluation
      response_format: options?.json ? { type: 'json_object' } : undefined
    });
  }
}
\end{lstlisting}
